\begin{frame}[plain]
  \vspace{0.6cm}
  {\fontsize{34}{38}\selectfont\bfseries Feature-Resolved Attention\par}
  \vspace{0.05cm}
  {\Large\color{muted}Pathway-specific steering with sparse SAE features inside attention heads\par}
  \vfill
  \TakehomeBox{Takeaway: FRA decomposes attention at the SAE feature-level, allowing for more finegrained attribution and intervention. We use it to find and remove localised sleeper triggers in the OV pathway, while distributed emergent misalignment needs broader QK-level intervention.}
  \vfill
  {\small\color{muted}QK resolves feature-pair interactions. OV resolves feature flow through frozen attention.}
\end{frame}

\MechanismFrame{1}{Plain Attention Head}{Start from the standard attention mechanism, with the residual stream feeding LayerNorm and receiving the head output.}

\MechanismFrame{2}{Add an SAE Basis}{The post-LayerNorm activation is expanded into sparse, interpretable SAE features.}

\MechanismFrame{3}{Zoom Into the Head}{Remove the residual-stream context and keep the same attention architecture fixed in the center.}

\MechanismFrame{4}{OV Pathway}{Hold the attention pattern fixed and trace which individual features are written through values.}

\MechanismFrame{5}{QK Interaction}{Trace feature pairs through query and key channels to resolve which features interact during a given behaviour.}

\begin{frame}{What FRA Scores}
  \framesubtitle{A feature-level decomposition for both sides of an attention head}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \SoftBox{qkFill}{ink}{%
        {\large\bfseries\color{qkC}QK asks where to attend}\par
        \vspace{0.16cm}
        \[
        \mathrm{FRA}^{\mathrm{QK},h}_{qk,\lambda,\mu}
        =
        \frac{f_q^\lambda f_k^\mu}{\sqrt{d_h}}\,
        W^{\mathrm{dec}}_\lambda W_{QK}^h
        (W^{\mathrm{dec}}_\mu)^\top .
        \]
        Feature pairs interact through the query-key score.
      }
    \column{0.48\textwidth}
      \SoftBox{ovFill}{ink}{%
        {\large\bfseries\color{ovC}OV asks what is written}\par
        \vspace{0.16cm}
        \[
        \mathrm{FRA}^{\mathrm{OV},h}_{qk,\lambda}
        =
        A^h_{qk} f_k^\lambda
        \left\lVert W^{\mathrm{dec}}_\lambda W_{OV}^h \right\rVert_2 .
        \]
        Individual features flow through values under fixed attention.
      }
  \end{columns}
  \vspace{0.35cm}
  \SoftBox{saeFill}{ink}{\bfseries Contribution: feature-level decomposition of both QK and OV attention circuits, followed by pathway-specific intervention.}
\end{frame}
