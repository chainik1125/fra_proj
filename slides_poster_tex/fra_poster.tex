\begin{tcolorbox}[sharp corners, boxrule=0.8pt, colframe=qkC, colback=qkFill, left=0.12in, right=0.12in]
\fontsize{30}{37}\selectfont
\textbf{\color{qkC}QK: feature interaction attribution + query/key intervention}
\[
\begin{aligned}
\mathrm{FRA}^{\mathrm{QK},h}_{qk,\lambda,\mu}
&=
\frac{f_q^\lambda f_k^\mu}{\sqrt{d_h}}\,
W^{\mathrm{dec}}_\lambda W_{QK}^h
(W^{\mathrm{dec}}_\mu)^\top \\[0.08in]
\widetilde{Q}^{h}_{q}
&= Q^{h}_{q} + \alpha f^\lambda_q W^{\mathrm{dec}}_\lambda W_Q^h,\\
\widetilde{K}^{h}_{k}
&= K^{h}_{k} + \alpha f^\mu_k W^{\mathrm{dec}}_\mu W_K^h
\end{aligned}
\]
\end{tcolorbox}

\begin{tcolorbox}[sharp corners, boxrule=0.8pt, colframe=ovC, colback=ovFill, left=0.12in, right=0.12in]
\fontsize{30}{37}\selectfont
\textbf{\color{ovC}OV: feature-to-residual write-strength attribution + value intervention}
\[
\begin{aligned}
\mathrm{FRA}^{\mathrm{OV},h}_{qk,\lambda}
&=
A^h_{qk} f_k^\lambda
\left\lVert W^{\mathrm{dec}}_\lambda W_{OV}^h \right\rVert_2 \\[0.08in]
\widetilde{V}^{h}_{k}
&= V^{h}_{k} + \alpha f^\lambda_k W^{\mathrm{dec}}_\lambda W_V^h
\end{aligned}
\]
\end{tcolorbox}
\end{posterbox}
