\section{Exact SAE reconstruction \& propagated OV error}

Using the same notation as defined in the appendix of the main paper, define the truncated SAE reconstruction at position $t$ by
\begin{equation}
    \hat{x}_t := \sum_{\lambda \in \mathcal A_t}f_t^\lambda W_\lambda^\dec + b^\dec \implies x_t = \sum_{\lambda \in \mathcal A_t}f_t^\lambda W_\lambda^\dec + b^\dec + \epsilon_t,
\end{equation}
where SAE reconstruction error is the residual vector $\epsilon_t := x_t - \hat{x}_t$, where $x_t$ is the true decomposition. Using the exact SAE reconstruction in the attention sublayer output at query position $q$, we have
\begin{align}
    r_q' &= r_q + \sum_h\sum_k A_{qk}^h \left(\sum_{\lambda \in \mathcal A_k}f_k^\lambda W_\lambda^\dec + b^\dec + \epsilon_k\right) W_{OV}^h \\
    &= r_q + \sum_h\sum_k A_{qk}^h \left[ \left(\sum_{\lambda \in \mathcal A_k}f_k^\lambda W_\lambda^\dec\right)W_{OV}^h + b^\dec W_{OV}^h + \epsilon_k W_{OV}^h \right] \\
    &= r_q + \sum_h\sum_k A_{qk}^h \left(\sum_{\lambda \in \mathcal A_k}f_k^\lambda W_\lambda^\dec\right)W_{OV}^h + \sum_h\sum_k A_{qk}^h b^\dec W_{OV}^h + \sum_h\sum_k A_{qk}^h \epsilon_k W_{OV}^h  \\
    &= r_q + \sum_h\sum_k \sum_{\lambda \in \mathcal A_k}A_{qk}^h f_k^\lambda W_\lambda^\dec W_{OV}^h + \underbrace{ \sum_h\sum_k A_{qk}^h b^\dec W_{OV}^h}_{b_q} + \underbrace{\sum_h\sum_k A_{qk}^h \epsilon_k W_{OV}^h}_{\Delta_q^{OV}}.
\end{align}
Here, we have defined $b_q :=\sum_h\sum_k A_{qk}^h b^\dec W_{OV}^h$ to be the propagated decoder-bias term at query position $q$ and $\Delta_q^{OV} := \sum_h\sum_k A_{qk}^h \epsilon_k W_{OV}^h$ is the propagated SAE reconstruction error at $q$. The exact feature-resolved OV decomposition is
\begin{equation}
    \boxed{r_q' = r_q + \sum_h\sum_k \sum_{\lambda \in \mathcal A_k}A_{qk}^h f_k^\lambda W_\lambda^\dec W_{OV}^h + b_q + \Delta_q^{OV}.}
\end{equation}

\section{OV error bound}
A useful norm bound immediately follows the above decomposition. Since each tensor entry $A_{qk}^h \geq 0$ and for each head $h$,  $\sum_k A_{qk}^h = 1$, we have
\[
\tnorm{\Delta_q^{OV}} = \tnorm{\sum_h \sum_k A_{qk}^h \epsilon_k W_{OV}^h} \leq \sum_h \sum_k A_{qk}^h \tnorm{\epsilon_k W_{OV}^h},
\]
via an application of the triangle inequality. Using the operator norm bound $\tnorm{\epsilon_k W_{OV}^h} \leq \tnorm{\epsilon_k}\opnorm{W_{OV}^h}$,
\begin{align}
\tnorm{\Delta_q^{OV}} = \tnorm{\sum_h \sum_k A_{qk}^h \epsilon_k W_{OV}^h} &\leq \sum_h \sum_k A_{qk}^h \tnorm{\epsilon_k W_{OV}^h} \\
&\leq \sum_h \sum_k A_{qk}^h \tnorm{\epsilon_k}\opnorm{W_{OV}^h} \\
&= \sum_h \opnorm{W_{OV}^h} \sum_k A_{qk}^h \tnorm{\epsilon_k} \\
&= \sum_h \opnorm{W_{OV}^h} \max_k \tnorm{\epsilon_k},
\end{align}
where in the final two lines we use that $W_{OV}^h$ does not depend on $k$ and $\sum_k A_{qk}^h = 1$. The final bound is
\begin{equation}
  \boxed{\tnorm{\Delta_q^{OV}} \leq  \sum_h \opnorm{W_{OV}^h} \max_k \tnorm{\epsilon_k}.}
\end{equation}
We have shown that the error in the OV decomposition is controlled by the SAE reconstruction error at the attended positions, weighted through the OV operator norms.

\section{Exact SAE reconstruction and propagated QK error}
We start from the exact SAE reconstruction introduced in the OV derivation:
\[
x_t = \sum_{\lambda \in \mathcal A_t}f_t^\lambda W_\lambda^\dec + b^\dec + \epsilon_t.
\]
Now consider attention head $h$ and the QK matrix
\begin{equation}
    W_{QK}^h := W_{Q}^h (W_K^h)^\top \in \mathbb{R}^{d_{\text{model}} \times d_{\text{model}}}.
\end{equation}
The pre-softmax attention score from query position $q$ to key position $k$ is
\begin{equation}
    s_{qk}^h = \frac{1}{\sqrt{d_{\text{head}}}} x_q W_{QK}^h x_k^\top.
\end{equation}
Using exact SAE decompositions:
\[
x_q = z_q + b^{\text{dec}} + \epsilon_q,\quad x_k = z_k + b^{\text{dec}} + \epsilon_k.
\]
Substituting this into the QK score:
\begin{align}
    s_{qk}^h &= \frac{1}{\sqrt{d_{\text{head}}}} (z_q + b^\dec + \epsilon_q) W_{QK}^h(z_k + b^\dec + \epsilon_k)^\top \\
    &= \frac{1}{\sqrt{d_{\text{head}}}} (z_q + b^\dec + \epsilon_q) W_{QK}^h(z_k^\top + (b^\dec)^\top + \epsilon_k^\top) \\
    &= \frac{1}{\sqrt{d_{\text{head}}}} \left[ (z_q + b^\dec + \epsilon_q)W_{QK}^h z_k^\top + (z_q + b^\dec + \epsilon_q)W_{QK} (b^\dec)^\top + (z_q + b^\dec + \epsilon_q)W_{QK} \epsilon_k^\top\right].
\end{align}
Distributing each term over the left-side sum gives us nine terms. We can expand the feature reconstruction components as $z_q = \sum_{\lambda \in \mathcal A_q}f_q^\lambda W_\lambda^\dec$ and $z_k = \sum_{\mu \in \mathcal A_k}f_k^\mu W_\mu^\dec$. The pure feature-feature term is
\begin{align}
    z_q W_{QK}^h z_k^\top &= \left( \sum_{\lambda \in \mathcal A_q}f_q^\lambda W_\lambda^\dec\right)W_{QK}^h\left(\sum_{\mu \in \mathcal A_k}f_k^\mu W_\mu^\dec\right)^\top = \sum_{\lambda \in \mathcal A_q}\sum_{\mu \in \mathcal A_k} f_q^\lambda f_k^\mu W_\lambda^\dec W_{QK}^h (W_\mu^\dec)^\top.
\end{align}
Therefore, the feature-feature contribution to the QK score is the original FRA-QK term:
\begin{equation}
    \sum_{\lambda \in \mathcal A_q}\sum_{\mu \in \mathcal A_k} \underbrace{\frac{f_q^\lambda f_k^\mu}{\sqrt{d_{\text{head}}}} W_\lambda^\dec W_{QK}^h (W_\mu^\dec)^\top}_{\mathrm{FRA}_{qk, \lambda, \mu}^{QK, h}}
\end{equation}
The remaining terms are exactly the terms involving the decoder bias and the SAE reconstruction error. We separate these into two pieces: a propagated decoder-bias contribution and a propagated reconstruction-error contribution. First, collect all terms involving $b^\dec$ but not involving $\epsilon_q$ or $\epsilon_k$:
\begin{equation}
    b_{qk}^{QK,h}
    :=
    \frac{1}{\sqrt{d_{\text{head}}}}
    \left[
        z_q W_{QK}^h (b^\dec)^\top
        +
        b^\dec W_{QK}^h z_k^\top
        +
        b^\dec W_{QK}^h (b^\dec)^\top
    \right].
\end{equation}
Expanding $z_q$ and $z_k$, this is equivalently
\begin{align}
    b_{qk}^{QK,h}
    =
    \frac{1}{\sqrt{d_{\text{head}}}}
    \bigg[
    \sum_{\lambda \in \mathcal A_q}
    f_q^\lambda
    W_\lambda^\dec
    W_{QK}^h
    (b^\dec)^\top
    +
    \sum_{\mu \in \mathcal A_k}
    f_k^\mu
    b^\dec
    W_{QK}^h
    (W_\mu^\dec)^\top
    +
    b^\dec
    W_{QK}^h
    (b^\dec)^\top
    \bigg].
\end{align}
This is the exact form of the $b_{qk}$ term in the main text, with the head index made explicit. Next, collect all terms involving the SAE reconstruction error. These are
\begin{align}
    \Delta_{qk}^{QK,h}
    :=
    \frac{1}{\sqrt{d_{\text{head}}}}
    \bigg[
    z_q W_{QK}^h \epsilon_k^\top
    +
    b^\dec W_{QK}^h \epsilon_k^\top
    +
    \epsilon_q W_{QK}^h z_k^\top
    +
    \epsilon_q W_{QK}^h (b^\dec)^\top
    +
    \epsilon_q W_{QK}^h \epsilon_k^\top
    \bigg].
\end{align}
Equivalently, grouping the non-error reconstruction terms together,
\begin{equation}
    \Delta_{qk}^{QK,h}
    =
    \frac{1}{\sqrt{d_{\text{head}}}}
    \left[
        (z_q + b^\dec)W_{QK}^h\epsilon_k^\top
        +
        \epsilon_q W_{QK}^h(z_k + b^\dec)^\top
        +
        \epsilon_q W_{QK}^h\epsilon_k^\top
    \right].
\end{equation}
Thus the exact QK decomposition is
\begin{equation}
\boxed{
    s_{qk}^h
    =
    \sum_{\lambda \in \mathcal A_q}
    \sum_{\mu \in \mathcal A_k}
    \mathrm{FRA}_{qk,\lambda,\mu}^{QK,h}
    +
    b_{qk}^{QK,h}
    +
    \Delta_{qk}^{QK,h}.
}
\end{equation}
Expanding the FRA term, this is
\begin{equation}
\boxed{
    s_{qk}^h
    =
    \sum_{\lambda \in \mathcal A_q}
    \sum_{\mu \in \mathcal A_k}
    \frac{f_q^\lambda f_k^\mu}{\sqrt{d_{\text{head}}}}
    W_\lambda^\dec
    W_{QK}^h
    (W_\mu^\dec)^\top
    +
    b_{qk}^{QK,h}
    +
    \Delta_{qk}^{QK,h}.
}
\end{equation}
In the limit of exact SAE reconstruction, $\epsilon_q = \epsilon_k = \Delta_{qk}^{QK,h} = 0$. So the decomposition reduces to
\begin{equation}
    s_{qk}^h
    =
    \sum_{\lambda \in \mathcal A_q}
    \sum_{\mu \in \mathcal A_k}
    \mathrm{FRA}_{qk,\lambda,\mu}^{QK,h}
    +
    b_{qk}^{QK,h}.
\end{equation}
This is the approximate QK decomposition from the main text, but written as an exact equality once the propagated SAE reconstruction error is included. The only additional complication relative to the OV case is that the QK score is bilinear in two reconstructed activations. In the OV path, the error propagates linearly through a single term of the form $\epsilon_k W_{OV}^h$. In the QK path, both the query-side error and key-side error contribute, producing feature-error, error-feature, and error-error terms:
\[
    (z_q+b^\dec)W_{QK}^h\epsilon_k^\top,
    \qquad
    \epsilon_q W_{QK}^h(z_k+b^\dec)^\top,
    \qquad
    \epsilon_q W_{QK}^h\epsilon_k^\top.
\]
No additional nonlinear correction is needed here because the decomposition is of the pre-softmax score $s_{qk}^h$, not of the post-softmax attention weight $A_{qk}^h$.
