    \pgfmathtruncatemacro{\keep}{(\r==0 && \c==0) || (\r==1 && \c==0) || (\r==2 && \c==1) || (\r==2 && \c==2) || (\r==3 && \c==0) || (\r==3 && \c==3) || (\r==4 && \c==1) || (\r==4 && \c==4) ? 1 : 0}%

    \pgfmathsetmacro{\xc}{(\c-2)*\HC}%
    \pgfmathsetmacro{\yc}{1.8 + (2-\r)*\HC}%
    \fill[qkC,opacity=\op]
      ([xshift=\xc mm-1mm,yshift=\yc mm-1mm]logits.center)
      rectangle
      ([xshift=\xc mm+1mm,yshift=\yc mm+1mm]logits.center);
  }%
}
