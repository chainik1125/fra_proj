\begin{posterbox}[colframe=saeC!45]{Method: Feature-Resolved Attention}
SAEs provide interpretable feature bases for residual streams and MLPs, but attention is still usually summarized by one weight per token pair.

\tagline{saeC}{FRA turns that opaque attention weight into sparse feature-level contributions.}

\includegraphics[width=\linewidth,height=9.4in,keepaspectratio]{figures/poster_fra_overview.pdf}


\begin{tcolorbox}[sharp corners, boxrule=0.8pt, colframe=qkC, colback=qkFill, left=0.12in, right=0.12in]
\fontsize{34}{42}\selectfont
\textbf{\color{qkC}QK asks where to attend}
\[
\mathrm{FRA}^{\mathrm{QK},h}_{qk,\lambda,\mu}
=
\frac{f_q^\lambda f_k^\mu}{\sqrt{d_h}}\,
W^{\mathrm{dec}}_\lambda W_{QK}^h
(W^{\mathrm{dec}}_\mu)^\top .
\]
\end{tcolorbox}

\begin{tcolorbox}[sharp corners, boxrule=0.8pt, colframe=ovC, colback=ovFill, left=0.12in, right=0.12in]
\fontsize{34}{42}\selectfont
\textbf{\color{ovC}OV asks what is written}
\[
\mathrm{FRA}^{\mathrm{OV},h}_{qk,\lambda}
=
A^h_{qk} f_k^\lambda
\left\lVert W^{\mathrm{dec}}_\lambda W_{OV}^h \right\rVert_2 .
\]
\end{tcolorbox}
\end{posterbox}

\begin{posterbox}[colframe=qkC!55]{Case Study 2: Emergent Misalignment}
\textbf{Model organism.} Qwen2.5-14B-Instruct LoRA variants induce harmful behavior across unrelated prompts.

\tagline{qkC}{Distributed behavior: QK-level intervention moves generation; single-layer OV is too narrow.}

\includegraphics[width=\linewidth,height=7.5in,keepaspectratio]{figures/phase1_fra_plus_additive_3domains_neg6.pdf}
\figcap{Headline $\Delta$ alignment at coherence $\geq 70$. QK$\to$QK is the strongest FRA recipe on finance and sports; conventional additive is strongest on medical.}

\vspace{0.08in}
\includegraphics[width=\linewidth,height=9.5in,keepaspectratio]{figures/phase1_2x3_seed42_neg6.pdf}
\figcap{Alignment-coherence frontiers show the same asymmetry: OV-only steering clusters near baseline, while QK$\to$QK shifts the reachable frontier.}

\vspace{0.10in}
\begin{tabular}{>{\bfseries}p{0.30\linewidth}p{0.31\linewidth}p{0.27\linewidth}}
\toprule
Domain & Best FRA & Best conventional \\
\midrule
Medical & QK$\to$QK: +25.0 & +39.4 \\
Finance & QK$\to$QK: +21.9 & +13.1 \\
Sports & QK$\to$QK: +37.7 & +28.3 \\
\bottomrule
\end{tabular}

\vspace{0.10in}
\textbf{Mechanistic read.} In GQA models, OV steering touches only a small value pathway. QK$\to$QK acts at the shared pre-attention representation, affecting query, key, and value computations across the layer.
\end{posterbox}

\begin{posterbox}[colframe=steerC!55]{Result Pattern}
\begin{tabular}{>{\bfseries}p{0.30\linewidth}p{0.62\linewidth}}
OV$\to$OV & Best when the target is localized and the attention pattern should stay fixed. \\
QK$\to$QK & Best FRA path for distributed EM behavior because it reaches the shared pre-attention representation. \\
Attribution & FRA-ranked QK features beat random features in the EM random-feature baseline. \\
\end{tabular}

\vspace{0.12in}
\textbf{Practical rule:} pick the narrowest pathway that reaches the behavior; widen only when the target is distributed across heads or layers.
